  endswith = 'endswith',
  contains = 'contains'
  // between = 'between',

export interface IExpression {
  var: string;
  op: SemTalkOperator | SemTalkAssignment;
  val: string;
}
export type ParsedJsonTaskConfig = {
  id: string;
  name: string;
  tasktype: BPMN_TaskTypeName;
  promptTemplate: string;
  cardPayload?: Record<string, any>;
  cardPayloadDefined: boolean;
  systemPrompt: string;
  systemPromptDefined: boolean;
  laneId?: string;
  model?: string;
  modelDefined: boolean;
  temperature?: number;
  temperatureDefined: boolean;
  toolNames: string[];
  toolFilterDefined: boolean;
  inputs: string[];
  outputs: string[];
  AssignmentExpression: IExpression[],
  WebserviceMethod?: string;
  WebserviceUrl?: string;
  WebserviceParams?: string;
  WebserviceHeaders?: string;
  WebserviceBody?: string;
};

export function parseJsonTaskConfig(
  task: any,
  _globalSystemPrompt: string,
  laneId?: string
) {
  const metadata = task?.attributes ?? {};

  const promptTemplate =
    toStringValue(metadata.AgenticPromptTemplate) ??
    `Perform task: ${task?.name ?? task?.id ?? "BPMN task"}`;

  const taskSystemPrompt = toStringValue(metadata.AgenticSystemPrompt);
  const systemPromptDefined = metadata.AgenticSystemPrompt !== undefined && metadata.AgenticSystemPrompt !== "";

  const cardPayloadValue = parseJsonValue(metadata.cardPayload ?? metadata.adaptiveCard ?? metadata.card);
  const cardPayloadDefined = metadata.Dialog;
  // metadata.cardPayload !== undefined ||
  // metadata.adaptiveCard !== undefined ||
  // metadata.card !== undefined;

  const modelDefined = metadata.AgenticModel !== undefined && metadata.AgenticModel !== "";
  const temperatureDefined = metadata.AgenticTemperature !== undefined && metadata.AgenticTemperature !== "";
  const toolFilterDefined = metadata.AgenticToolnames !== undefined && metadata.AgenticToolnames !== "";
  const assignmentExpression: IExpression[] = metadata.AssignmentExpression != undefined ? JSON.parse(metadata.AssignmentExpression) : [];

  return {
    id: String(task?.id ?? ""),
    name: String(task?.name ?? task?.id ?? "BPMN task"),
    tasktype: task?.tasktype ?? BPMN_TaskType.None,
    promptTemplate,
    cardPayload: cardPayloadValue,
    cardPayloadDefined,
    systemPrompt: taskSystemPrompt ?? "",
    systemPromptDefined,
    laneId,
    model: toStringValue(metadata.AgenticModel),
    modelDefined,
    temperature: parseTemperature(metadata.AgenticTemperature),
    temperatureDefined,
    toolNames: toStringList(metadata.AgenticToolnames ?? metadata.toolNames ?? metadata.toolList),
    toolFilterDefined,
    inputs: task.inputs,
    outputs: task.outputs,
    AssignmentExpression: assignmentExpression,
    WebserviceMethod: toStringValue(metadata.WebserviceMethod),
    WebserviceUrl: toStringValue(metadata.WebserviceUrl),
    WebserviceParams: toStringValue(metadata.WebserviceParams),
    WebserviceHeaders: toStringValue(metadata.WebserviceHeaders),
    WebserviceBody: toStringValue(metadata.WebserviceBody),
  } satisfies ParsedJsonTaskConfig;
}

function parseJsonValue(value: unknown) {
  if (value === undefined || value === null || value === "") {
    return undefined;
  }

  if (typeof value === "string") {
    try {
      return JSON.parse(value);
    } catch {
      return undefined;
    }
  }

  if (typeof value === "object") {
    return value as Record<string, any>;
  }

  return undefined;
}
